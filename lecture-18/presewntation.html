01 · JWT

რა არის JWT?

JSON Web Token

კომპაქტური, URL-უსაფრთხო ტოკენი
სახელი → პრეტენზიების (claims) გადაცემა

სტრუქტურა: 3 ნაწილი

header.payload.signature
● Header — ალგორითმი და ტიპი
● Payload — claims (sub, exp, iat…)
● Signature — header + payload + secret

04 · GET

ტიპიზებული fetch GET

// api/users.ts
import { User } from "../types";

export async function getUsers(): Promise<User[]> {
  const token = localStorage.getItem("token");

  const res = await fetch("/api/users", {
    method: "GET",
    headers: {
      "Content-Type": "application/json",
      "Authorization": `Bearer ${token}`,
    },
  });

  if (!res.ok) throw new Error("Failed to fetch");

  // ტიპიზებული პარსინგი
  const data: User[] = await res.json();
  return data;
}

05 · POST

ტიპიზებული fetch POST

// api/auth.ts
import { LoginRequest, LoginResponse } from "../types";

export async function login(
  body: LoginRequest
): Promise<LoginResponse> {
  const res = await fetch("/api/auth/login", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(body),
  });

  if (!res.ok) {
    const err = await res.json();
    throw new Error(err.message);
  }
  return res.json();
}
// გამოძახება
const result = await login({
  email: "user@mail.com",
  password: "secret123",
});

localStorage.setItem("token", result.accessToken);

// result.user — სრულად ტიპიზებულია
console.log(result.user.name);
console.log(result.user.role);

  name: string;
  price: number;
  category: "electronics" | "books" | "clothing";
  tags?: string[];
}

export interface Product extends CreateProductDto {
  id: string;
  createdAt: string;
}
// api/products.ts
export async function createProduct(
  dto: CreateProductDto
): Promise<Product> {
  const token = localStorage.getItem("token");
  const res = await fetch("/api/products", {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "Authorization": `Bearer ${token}`,
    },
    body: JSON.stringify(dto),
  });
  return res.json();
}

07 · Binding

მონაცემების ბაინდინგი კომპონენტებში

// components/UserList.tsx
import { useEffect, useState } from "react";
import { User } from "../types";
import { getUsers } from "../api/users";

export function UserList() {
  const [users, setUsers] = useState<User[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    getUsers().then(setUsers).finally(() =>
      setLoading(false)
    );
  }, []);

  if (loading) return <p>იტვირთება...</p>;

  return (
    <ul>
      {users.map((u) => (
        <li key={u.id}>{u.name} — {u.role}</li>
      ))}
    </ul>
  );
}

// layouts/AppShell.tsx
interface AppShellProps {
  header: ReactNode;
  sidebar: ReactNode;
  footer?: ReactNode;
  children: ReactNode; // main slot
}

export function AppShell({
  header, sidebar, footer, children
}: AppShellProps) {
  return (
    <div className="grid grid-cols-[240px_1fr] grid-rows-[auto_1fr_auto] h-screen">
      <header className="col-span-2">{header}</header>
      <aside>{sidebar}</aside>
      <main>{children}</main>
      {footer && <footer className="col-span-2">{footer}</footer>}
    </div>
  );
}

10 · Practice

ყველაფერი ერთად: Protected Route

// components/ProtectedRoute.tsx
import { ReactNode } from "react";
import { Navigate } from "react-router-dom";

interface Props {
  children: ReactNode;
}

export function ProtectedRoute({ children }: Props) {
  const token = localStorage.getItem("token");

  if (!token) return <Navigate to="/login" />;
  return <>{children}</>;
}
// App.tsx
<Routes>
  <Route path="/login" element={<LoginPage />} />

  <Route path="/" element={
    <ProtectedRoute>
      <MainLayout>
        <Dashboard />
      </MainLayout>
    </ProtectedRoute>
  } />
</Routes>
